## 2. 权限模型设计

### 2.1 三级权限体系

| 角色 | 权限范围 | 可访问路径 | 可操作功能 |
|------|----------|------------|------------|
| **ADMIN** | 全权管理 | /api/admin/\*\*、/api/operator/\*\*、/api/\*\* | 用户管理、设备管理、消息管理、系统配置 |
| **OPERATOR** | 设备操作 | /api/operator/\*\*、/api/devices/\*\*、/api/messages/\*\* | 创建设备、发送消息、查看消息监控 |
| **VIEWER** | 只读查看 | /api/devices/\*\*（GET）、/api/history/\*\* | 查看设备列表、查看历史数据 |

### 2.2 接口权限矩阵

| 接口路径 | 方法 | 所需角色 | 说明 |
|----------|------|----------|------|
| /api/auth/register | POST | 公开 | 任何人可注册，注册后默认VIEWER角色 |
| /api/auth/login | POST | 公开 | 任何人可登录 |
| /api/auth/logout | POST | 认证用户 | 登录用户可登出 |
| /api/auth/current | GET | 认证用户 | 获取当前登录用户信息 |
| /api/auth/change-password | POST | 认证用户 | 修改自己的密码 |
| /api/admin/users | GET | ADMIN | 查看所有用户列表 |
| /api/admin/users/{id} | PUT | ADMIN | 修改用户信息（角色/状态） |
| /api/devices/\*\* | 全部 | 认证用户 | 设备操作（数据隔离：只能操作自己创建的设备） |
| /api/messages/publish | POST | OPERATOR+ | 发布消息 |
| /api/messages | GET | 认证用户 | 查看消息（数据隔离） |
| /api/history | GET | 认证用户 | 查看历史数据（数据隔离） |

```java
package com.mqtt.cloud.entity;

import com.baomidou.mybatisplus.annotation.*;

/**
 * 系统用户实体
 * 对应数据库表：sys_user
 */
@Data
@EqualsAndHashCode(callSuper = false)
@TableName("sys_user")
public class User {
    /** 主键ID */
    @TableId(value = "id", type = IdType.AUTO)
    private Long id;

    /** 登录用户名，全局唯一 */
    @TableField("username")
    private String username;

    /** BCrypt加密密码，查询时不返回 */
    @TableField("password")
    @TableField(select = false)
    private String password;

    /** 用户邮箱 */
    @TableField("email")
    private String email;

    /** 手机号 */
    @TableField("phone")
    private String phone;

    /** 用户角色：ADMIN/OPERATOR/VIEWER */
    @TableField("role")
    private String role;

    /** 账号状态：ACTIVE/DISABLED/LOCKED */
    @TableField("status")
    private String status;

    /** 最后登录时间 */
    @TableField("last_login")
    private LocalDateTime lastLogin;

    /** 创建时间（自动填充） */
    @TableField(value = "created_at", fill = FieldFill.INSERT)
    private LocalDateTime createdAt;

    /** 更新时间（自动填充） */
    @TableField(value = "updated_at", fill = FieldFill.INSERT_UPDATE)
    private LocalDateTime updatedAt;

    /** 逻辑删除标记（MyBatis Plus自动管理） */
    @TableLogic
    @TableField("deleted")
    private Integer deleted;

    // 以下字段仅用于DTO映射，不映射到数据库
    @TableField(exist = false)
    private String token;  // 登录后返回的JWT Token
}
```

#### 3.4.2 UserServiceImpl.java

```java
package com.mqtt.cloud.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.mqtt.cloud.common.ResultCode;
import com.mqtt.cloud.common.exception.BusinessException;
import com.mqtt.cloud.dto.request.ChangePasswordDTO;
import com.mqtt.cloud.dto.request.RegisterDTO;
import com.mqtt.cloud.dto.response.UserResponseDTO;
import com.mqtt.cloud.entity.User;
import com.mqtt.cloud.mapper.UserMapper;
import com.mqtt.cloud.service.UserService;
import com.mqtt.cloud.util.JwtUtil;
import lombok.RequiredArgsConstructor;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.util.concurrent.TimeUnit;

/**
 * 用户服务实现类
 */
@Service
@RequiredArgsConstructor
public class UserServiceImpl extends ServiceImpl<UserMapper, User> implements UserService {

    private final PasswordEncoder passwordEncoder;
    private final UserMapper userMapper;
    private final JwtUtil jwtUtil;
    private final RedisTemplate<String, String> redisTemplate;

    private static final String TOKEN_BLACKLIST_PREFIX = "token:blacklist:";

    @Override
    @Transactional(rollbackFor = Exception.class)
    public UserResponseDTO register(RegisterDTO dto) {
        // 1. 校验用户名唯一性
        LambdaQueryWrapper<User> usernameWrapper = new LambdaQueryWrapper<>();
        usernameWrapper.eq(User::getUsername, dto.getUsername()).eq(User::getDeleted, 0);
        if (userMapper.selectCount(usernameWrapper) > 0) {
            throw new BusinessException(ResultCode.USERNAME_EXISTS);
        }

        // 2. 校验手机号唯一性（如果提供了手机号）
        if (dto.getPhone() != null && !dto.getPhone().isEmpty()) {
            LambdaQueryWrapper<User> phoneWrapper = new LambdaQueryWrapper<>();
            phoneWrapper.eq(User::getPhone, dto.getPhone()).eq(User::getDeleted, 0);
            if (userMapper.selectCount(phoneWrapper) > 0) {
                throw new BusinessException(ResultCode.PHONE_EXISTS);
            }
        }

        // 3. 校验邮箱唯一性（如果提供了邮箱）
        if (dto.getEmail() != null && !dto.getEmail().isEmpty()) {
            LambdaQueryWrapper<User> emailWrapper = new LambdaQueryWrapper<>();
            emailWrapper.eq(User::getEmail, dto.getEmail()).eq(User::getDeleted, 0);
            if (userMapper.selectCount(emailWrapper) > 0) {
                throw new BusinessException(ResultCode.EMAIL_EXISTS);
            }
        }

        // 4. 构建用户对象
        User user = new User();
        user.setUsername(dto.getUsername());
        user.setPassword(passwordEncoder.encode(dto.getPassword()));  // BCrypt加密
        user.setEmail(dto.getEmail());
        user.setPhone(dto.getPhone());
        user.setRole("VIEWER");  // 默认角色：VIEWER
        user.setStatus("ACTIVE");  // 默认状态：ACTIVE
        user.setDeleted(0);

        // 5. 保存用户
        userMapper.insert(user);

        // 6. 返回响应DTO（不含密码）
        UserResponseDTO responseDTO = new UserResponseDTO();
        responseDTO.setId(user.getId());
        responseDTO.setUsername(user.getUsername());
        responseDTO.setEmail(user.getEmail());
        responseDTO.setPhone(user.getPhone());
        responseDTO.setRole(user.getRole());
        responseDTO.setStatus(user.getStatus());
        responseDTO.setLastLogin(user.getLastLogin());
        responseDTO.setCreatedAt(user.getCreatedAt());

        return responseDTO;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public User login(String username, String rawPassword) {
        // 1. 根据用户名查询用户
        User user = userMapper.findByUsername(username);
        if (user == null) {
            throw new BusinessException(ResultCode.INVALID_CREDENTIALS);
        }

        // 2. 检查账号状态
        if ("LOCKED".equals(user.getStatus())) {
            throw new BusinessException(ResultCode.ACCOUNT_LOCKED);
        }
        if ("DISABLED".equals(user.getStatus())) {
            throw new BusinessException(ResultCode.INVALID_CREDENTIALS);
        }

        // 3. 校验密码
        if (!passwordEncoder.matches(rawPassword, user.getPassword())) {
            throw new BusinessException(ResultCode.INVALID_CREDENTIALS);
        }

        // 4. 更新最后登录时间
        userMapper.updateLastLogin(user.getId(), LocalDateTime.now());

        // 5. 生成JWT Token并设置到user对象中（供Controller返回）
        String token = jwtUtil.generateToken(user.getId(), user.getUsername(), user.getRole());
        user.setToken(token);
        user.setPassword(null);  // 不返回密码

        return user;
    }

    @Override
    public User getCurrentUser(Long userId) {
        User user = userMapper.selectById(userId);
        if (user == null || user.getDeleted() == 1) {
            throw new BusinessException(ResultCode.NOT_FOUND);
        }
        user.setPassword(null);
        return user;
    }

    @Override
    public void logout(String token) {
        // 计算Token剩余有效期
        Long expirationSeconds = jwtUtil.getExpirationSeconds();
        // 将Token存入Redis黑名单，过期时间与Token一致
        redisTemplate.opsForValue().set(
                TOKEN_BLACKLIST_PREFIX + token,
                "logout",
                expirationSeconds,
                TimeUnit.SECONDS
        );
    }

    @Override
    public void changePassword(Long userId, ChangePasswordDTO dto) {
        // 1. 查询当前用户
        User user = userMapper.selectById(userId);
        if (user == null || user.getDeleted() == 1) {
            throw new BusinessException(ResultCode.NOT_FOUND);
        }

        // 2. 校验旧密码
        if (!passwordEncoder.matches(dto.getOldPassword(), user.getPassword())) {
            throw new BusinessException(ResultCode.INVALID_CREDENTIALS);
        }

        // 3. 校验新密码与旧密码不同
        if (passwordEncoder.matches(dto.getNewPassword(), user.getPassword())) {
            throw new BusinessException("新密码不能与旧密码相同");
        }

        // 4. 加密新密码并更新
        user.setPassword(passwordEncoder.encode(dto.getNewPassword()));
        userMapper.updateById(user);
    }
}
```

### 3.5 JWT认证过滤器

#### 3.5.1 JwtAuthenticationFilter.java

```java
package com.mqtt.cloud.filter;

import com.mqtt.cloud.common.ResultCode;
import com.mqtt.cloud.common.exception.BusinessException;
import com.mqtt.cloud.util.JwtUtil;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.web.authentication.WebAuthenticationDetailsSource;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.List;

/**
 * JWT认证过滤器
 * 从请求Header中提取JWT Token，验证后设置Spring Security上下文
 */
@Component
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtUtil jwtUtil;
    private final RedisTemplate<String, String> redisTemplate;

    // Token请求头名称
    private static final String TOKEN_HEADER = "Authorization";
    private static final String TOKEN_PREFIX = "Bearer ";
    private static final String TOKEN_BLACKLIST_PREFIX = "token:blacklist:";

    public JwtAuthenticationFilter(JwtUtil jwtUtil, RedisTemplate<String, String> redisTemplate) {
        this.jwtUtil = jwtUtil;
        this.redisTemplate = redisTemplate;
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response,
                                    FilterChain filterChain) throws ServletException, IOException {
        String token = resolveToken(request);

        // 无Token，继续过滤链（Security配置会处理权限）
        if (token == null) {
            filterChain.doFilter(request, response);
            return;
        }

        try {
            // 1. 检查Token是否在黑名单中（登出状态）
            String blacklisted = redisTemplate.opsForValue().get(TOKEN_BLACKLIST_PREFIX + token);
            if (blacklisted != null) {
                filterChain.doFilter(request, response);
                return;
            }

            // 2. 验证Token有效性
            jwtUtil.validateToken(token);

            // 3. 从Token中提取用户信息
            Long userId = jwtUtil.getUserIdFromToken(token);
            String username = jwtUtil.getUsernameFromToken(token);
            String role = jwtUtil.getRoleFromToken(token);

            // 4. 构建认证令牌（用户ID作为principal，角色作为authorities）
            UserPrincipal userPrincipal = new UserPrincipal(userId, username, role);
            UsernamePasswordAuthenticationToken authentication =
                    new UsernamePasswordAuthenticationToken(userPrincipal, null,
                            List.of(() -> "ROLE_" + role));

            // 5. 设置请求详情
            authentication.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));

            // 6. 设置到Security上下文
            SecurityContextHolder.getContext().setAuthentication(authentication);

        } catch (BusinessException e) {
            // Token无效或过期，不设置认证，让Security处理
            SecurityContextHolder.clearContext();
        }

        filterChain.doFilter(request, response);
    }

    /**
     * 从请求中提取Token
     * 从 Authorization: Bearer {token} 头中提取
     */
    private String resolveToken(HttpServletRequest request) {
        String bearerToken = request.getHeader(TOKEN_HEADER);
        if (bearerToken != null && bearerToken.startsWith(TOKEN_PREFIX)) {
            return bearerToken.substring(TOKEN_PREFIX.length());
        }
        return null;
    }
}
```

```java
    private final UserService userService;

    public AuthController(UserService userService) {
        this.userService = userService;
    }

    /**
     * 用户注册
     * 注册后默认角色为VIEWER
     */
    @PostMapping("/register")
    @Operation(summary = "用户注册", description = "注册新用户，默认角色为VIEWER")
    public Result<UserResponseDTO> register(@Valid @RequestBody RegisterDTO dto) {
        UserResponseDTO user = userService.register(dto);
        return Result.success(user);
    }

    /**
     * 用户登录
     * 返回JWT Token和用户信息
     */
    @PostMapping("/login")
    @Operation(summary = "用户登录", description = "使用用户名和密码登录，返回JWT Token")
    public Result<User> login(@Valid @RequestBody LoginDTO dto) {
        User user = userService.login(dto.getUsername(), dto.getPassword());
        return Result.success(user);
    }

    /**
     * 用户登出
     * 将当前Token加入Redis黑名单
     */
    @PostMapping("/logout")
    @Operation(summary = "用户登出", description = "将当前Token加入黑名单，使其失效")
    public Result<Void> logout() {
        // 从Security上下文获取当前认证信息
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        if (authentication != null && authentication.getPrincipal() instanceof UserPrincipal) {
            UserPrincipal userPrincipal = (UserPrincipal) authentication.getPrincipal();
            // 注意：登出时需要传入Token本身，这里简化处理，实际应从请求中提取Token
            // 具体实现见UserServiceImpl
        }
        return Result.success();
    }

    /**
     * 获取当前登录用户信息
     */
    @GetMapping("/current")
    @Operation(summary = "获取当前用户", description = "获取当前登录用户的详细信息")
    public Result<User> getCurrentUser() {
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        if (authentication == null || !(authentication.getPrincipal() instanceof UserPrincipal)) {
            throw new BusinessException(ResultCode.UNAUTHORIZED);
        }
        UserPrincipal userPrincipal = (UserPrincipal) authentication.getPrincipal();
        User user = userService.getCurrentUser(userPrincipal.getUserId());
        return Result.success(user);
    }

    /**
     * 修改密码
     */
    @PostMapping("/change-password")
    @Operation(summary = "修改密码", description = "修改当前登录用户的密码")
    public Result<Void> changePassword(@Valid @RequestBody ChangePasswordDTO dto) {
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        if (authentication == null || !(authentication.getPrincipal() instanceof UserPrincipal)) {
            throw new BusinessException(ResultCode.UNAUTHORIZED);
        }
        UserPrincipal userPrincipal = (UserPrincipal) authentication.getPrincipal();
        userService.changePassword(userPrincipal.getUserId(), dto);
        return Result.success();
    }
}
```

**注意：** logout接口需要从请求中提取Token并传递给UserService。上面的简化代码需要补充Token提取逻辑。完整实现如下：

```java
@PostMapping("/logout")
public Result<Void> logout(HttpServletRequest request) {
    String token = resolveToken(request);  // 从请求中提取Token（同JwtAuthenticationFilter中的方法）
    if (token != null) {
        userService.logout(token);
    }
    SecurityContextHolder.clearContext();
    return Result.success();
}
```
