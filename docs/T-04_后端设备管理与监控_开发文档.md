## 2. 设备状态流转设计

### 2.1 状态机

```
                    ┌─────────────────────────────────────────┐
                    │                                         │
                    ▼                                         │
              ┌──────────┐                                  │
              │ INACTIVE │ ◄──── 设备被删除后从列表移除      │
              └────┬─────┘                                  │
                   │ 创建设备后默认状态                      │
                   │ 或设备被删除/重置后                     │
                   ▼                                        │
              ┌──────────┐  收到MQTT心跳/数据消息            │
              │ OFFLINE  │ ──────────────────────► ONLINE   │
              └────┬─────┘   （lastSeen更新为当前时间）      │
                   │                                        │
                   │ 超过5分钟未收到消息                     │
                   │ （定时任务扫描检测）                     │
                   │                                        │
                   │ 设备主动发送离线消息                     │
                   │ （MQTT Last Will Testament）            │
                   │                                        │
                   └──────────────────────────────────────► │
                                                             │
                    ┌─────────────────────────────────────────┘
                    │
                    │ 设备被删除（级联删除）
                    ▼
              ┌──────────┐
              │ DELETED  │
              └──────────┘
```

**状态流转规则：**

| 触发条件 | 原状态 | 新状态 | 触发方式 |
|----------|--------|--------|----------|
| 创建设备 | - | INACTIVE | 调用创建设备API |
| 收到设备MQTT消息 | INACTIVE/OFFLINE | ONLINE | MqttMessageHandler处理 |
| 超过5分钟未收到消息 | ONLINE | OFFLINE | 定时任务扫描 |
| 设备发送LWT离线消息 | ONLINE | OFFLINE | MQTT Last Will Testament |
| 删除设备 | 任意 | DELETED | 调用删除API（级联删除） |

### 2.2 MQTT Topic约定

| Topic模式 | 用途 | QoS | 方向 |
|-----------|------|-----|------|
| `device/{device_key}/data` | 设备上报数据 | 0/1 | 设备→平台 |
| `device/{device_key}/command` | 平台下发命令 | 1 | 平台→设备 |
| `device/{device_key}/status` | 设备上报状态 | 1 | 设备→平台 |
| `device/{device_key}/heartbeat` | 设备心跳 | 0 | 设备→平台 |
| `device/{device_key}/lwt` | 设备遗嘱消息 | 1 | 设备→平台 |

#### 3.4.2 DeviceServiceImpl.java

```java
package com.mqtt.cloud.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.core.metadata.IPage;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.mqtt.cloud.common.ResultCode;
import com.mqtt.cloud.common.exception.BusinessException;
import com.mqtt.cloud.dto.request.CreateDeviceDTO;
import com.mqtt.cloud.dto.request.DeviceQueryDTO;
import com.mqtt.cloud.dto.request.UpdateDeviceDTO;
import com.mqtt.cloud.entity.Device;
import com.mqtt.cloud.entity.DeviceStatus;
import com.mqtt.cloud.mapper.DeviceMapper;
import com.mqtt.cloud.service.DeviceService;
import com.mqtt.cloud.service.DeviceStatusHistoryService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;

/**
 * 设备服务实现类
 */
@Service
@RequiredArgsConstructor
public class DeviceServiceImpl extends ServiceImpl<DeviceMapper, Device> implements DeviceService {

    private final DeviceStatusHistoryService deviceStatusHistoryService;

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Device createDevice(Long userId, CreateDeviceDTO dto) {
        // 1. 校验deviceKey唯一性
        LambdaQueryWrapper<Device> keyWrapper = new LambdaQueryWrapper<>();
        keyWrapper.eq(Device::getDeviceKey, dto.getDeviceKey()).eq(Device::getDeleted, 0);
        if (this.count(keyWrapper) > 0) {
            throw new BusinessException(ResultCode.DEVICE_KEY_EXISTS);
        }

        // 2. 构建设备对象
        Device device = new Device();
        device.setDeviceName(dto.getDeviceName());
        device.setDeviceKey(dto.getDeviceKey());
        device.setDeviceType(dto.getDeviceType());
        device.setTopic(dto.getTopic());
        device.setDescription(dto.getDescription());
        device.setOwnerId(userId);
        device.setStatus("INACTIVE");  // 初始状态：未激活
        device.setLastSeen(null);
        device.setDeleted(0);

        // 3. 保存设备
        this.save(device);

        return device;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Device updateDevice(Long deviceId, UpdateDeviceDTO dto) {
        // 1. 查询设备
        Device device = this.getById(deviceId);
        if (device == null || device.getDeleted() == 1) {
            throw new BusinessException(ResultCode.DEVICE_NOT_FOUND);
        }

        // 2. 校验权限（只能更新自己创建的设备）
        // 权限校验在Controller层通过@PreAuthorize或手动校验完成

        // 3. 更新字段
        if (dto.getDeviceName() != null) {
            device.setDeviceName(dto.getDeviceName());
        }
        if (dto.getDeviceType() != null) {
            device.setDeviceType(dto.getDeviceType());
        }
        if (dto.getTopic() != null) {
            device.setTopic(dto.getTopic());
        }
        if (dto.getDescription() != null) {
            device.setDescription(dto.getDescription());
        }

        this.updateById(device);
        return device;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void deleteDevice(Long deviceId) {
        Device device = this.getById(deviceId);
        if (device == null || device.getDeleted() == 1) {
            throw new BusinessException(ResultCode.DEVICE_NOT_FOUND);
        }

        // 逻辑删除
        device.setDeleted(1);
        this.updateById(device);

        // 设备状态标记为OFFLINE（避免状态监控任务继续扫描）
        device.setStatus("OFFLINE");
        this.updateById(device);
    }

    @Override
    public IPage<Device> getDevices(Long userId, DeviceQueryDTO dto) {
        Page<Device> page = new Page<>(dto.getPageNum(), dto.getPageSize());
        return this.baseMapper.pageQuery(page, dto, userId);
    }

    @Override
    public Device getDeviceById(Long deviceId) {
        Device device = this.getById(deviceId);
        if (device == null || device.getDeleted() == 1) {
            throw new BusinessException(ResultCode.DEVICE_NOT_FOUND);
        }
        return device;
    }

    @Override
    public Device getDeviceByKey(String deviceKey) {
        Device device = this.baseMapper.findByDeviceKey(deviceKey);
        if (device == null || device.getDeleted() == 1) {
            throw new BusinessException(ResultCode.DEVICE_NOT_FOUND);
        }
        return device;
    }

    @Override
    public List<Device> getOnlineDevices() {
        return this.baseMapper.findOnlineDevices();
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void updateDeviceStatus(Long deviceId, String status) {
        Device device = this.getById(deviceId);
        if (device == null) return;

        String oldStatus = device.getStatus();
        if (oldStatus.equals(status)) return;  // 状态未变化，不处理

        // 更新设备状态
        device.setStatus(status);
        if ("ONLINE".equals(status)) {
            device.setLastSeen(LocalDateTime.now());
        }
        this.updateById(device);

        // 记录状态变更历史
        DeviceStatus history = new DeviceStatus();
        history.setDeviceId(deviceId);
        history.setStatus(status);
        history.setTimestamp(LocalDateTime.now());
        // 通过DeviceStatusHistoryService保存
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void updateDeviceStatusByKey(String deviceKey, String status) {
        Device device = this.baseMapper.findByDeviceKey(deviceKey);
        if (device == null) return;

        updateDeviceStatus(device.getId(), status);
    }
}
```

```java
package com.mqtt.cloud.service;

import com.baomidou.mybatisplus.extension.service.IService;
```

```java
import org.springframework.scheduling.annotation.Scheduled;

/**
 * 设备监控服务
 * 定时检查设备在线状态
 */
@Service
public class DeviceMonitorService {

    private final DeviceService deviceService;
    private final DeviceStatusHistoryService deviceStatusHistoryService;

    // 超时时间：5分钟（从application.yml读取）
    private final int STATUS_TIMEOUT_MINUTES = 5;

    public DeviceMonitorService(DeviceService deviceService,
                                 DeviceStatusHistoryService deviceStatusHistoryService) {
        this.deviceService = deviceService;
        this.deviceStatusHistoryService = deviceStatusHistoryService;
    }

    /**

     * 超过5分钟未上报的设备标记为OFFLINE
     */
    @Scheduled(fixedRateString = "${app.device.status-check-interval:60000}")
    public void checkDeviceOnlineStatus() {
        // 1. 查找超时设备（状态为ONLINE但超过5分钟未上报）
        List<Device> timeoutDevices = deviceService
                .getBaseMapper()
                .findTimeoutDevices(STATUS_TIMEOUT_MINUTES);

        // 2. 将超时设备标记为OFFLINE
        for (Device device : timeoutDevices) {
            deviceService.updateDeviceStatus(device.getId(), "OFFLINE");
            deviceStatusHistoryService.recordStatusChange(device.getId(), "OFFLINE");
        }

        if (!timeoutDevices.isEmpty()) {
            // 记录日志（实际项目应使用SLF4J）
            System.out.println("设备状态检查完成，发现 " + timeoutDevices.size() + " 个设备超时离线");
        }
```

```java
package com.mqtt.cloud.mqtt;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.mqtt.cloud.entity.Device;
import com.mqtt.cloud.entity.DeviceStatus;
import com.mqtt.cloud.service.DeviceService;
import com.mqtt.cloud.service.DeviceStatusHistoryService;
import com.mqtt.cloud.service.MessageService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.eclipse.paho.client.mqttv3.*;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

/**
 * MQTT消息处理器
 * 处理设备上报的消息，更新设备状态，保存消息记录
 */
@Component
@RequiredArgsConstructor
@Slf4j
public class MqttMessageHandler implements MqttCallback {

    private final DeviceService deviceService;
    private final MessageService messageService;
    private final DeviceStatusHistoryService deviceStatusHistoryService;
    private final ObjectMapper objectMapper;

    // MQTT配置
    @Value("${spring.mqtt.username}")
    private String mqttUsername;

    @Value("${spring.mqtt.password}")
    private String mqttPassword;

    @Value("${spring.mqtt.clientId}")
    private String mqttClientId;

    @Value("${spring.mqtt.host}")
    private String mqttHost;

    private MqttClient mqttClient;

    /**
     * 初始化MQTT客户端并订阅所有设备Topic
     * 在应用启动时调用
     */
    public void init() {
        try {
            // 构建MQTT客户端
            String clientId = mqttClientId + "_handler_" + System.currentTimeMillis();
            mqttClient = new MqttClient(mqttHost, clientId, new MemoryPersistence());

            MqttConnectOptions options = new MqttConnectOptions();
            options.setUserName(mqttUsername);
            options.setPassword(mqttPassword.toCharArray());
            options.setCleanSession(true);
            options.setAutomaticReconnect(true);  // 自动重连
            options.setConnectionTimeout(10);

            // 设置回调
            mqttClient.setCallback(this);

            // 连接Broker
            mqttClient.connect(options);
            log.info("MQTT消息处理器连接成功：{}", mqttHost);

            // 订阅所有设备Topic（使用通配符）
            mqttClient.subscribe("device/+/data", 1);
            mqttClient.subscribe("device/+/heartbeat", 0);
            mqttClient.subscribe("device/+/lwt", 1);
            log.info("MQTT Topic订阅成功：device/+/data, device/+/heartbeat, device/+/lwt");

        } catch (MqttException e) {
            log.error("MQTT消息处理器初始化失败", e);
        }
    }

    /**
     * 收到消息回调
     */
    @Override
    public void messageArrived(String topic, MqttMessage mqttMessage) {
        String payload = new String(mqttMessage.getMessage());
        log.debug("收到MQTT消息: topic={}, payload={}", topic, payload);

        // 1. 从Topic中提取device_key
        // Topic格式：device/{device_key}/{type}
        String[] topicParts = topic.split("/");
        if (topicParts.length < 3) {
            log.warn("Topic格式不正确: {}", topic);
            return;
        }
        String deviceKey = topicParts[1];
        String messageType = topicParts[2];  // data/heartbeat/lwt

        try {
            // 2. 根据device_key查找设备
            Device device = deviceService.getDeviceByKey(deviceKey);

            // 3. 更新设备lastSeen和状态
            if ("data".equals(messageType) || "heartbeat".equals(messageType)) {
                deviceService.updateDeviceStatusByKey(deviceKey, "ONLINE");
                deviceStatusHistoryService.recordStatusChange(device.getId(), "ONLINE");
            } else if ("lwt".equals(messageType)) {
                // LWT消息：设备异常离线
                deviceService.updateDeviceStatusByKey(deviceKey, "OFFLINE");
                deviceStatusHistoryService.recordStatusChange(device.getId(), "OFFLINE");
            }

            // 4. 保存消息记录
            messageService.saveReceivedMessage(
                    device.getId(),
                    topic,
                    payload,
                    mqttMessage.getQos(),
                    "SUBSCRIBE"
            );

        } catch (BusinessException e) {
            // 设备不存在，忽略该消息
            log.warn("收到未知设备的消息: device_key={}, topic={}", deviceKey, topic);
        } catch (Exception e) {
            log.error("处理MQTT消息失败: topic={}", topic, e);
        }
    }

    /**
     * 连接丢失回调
     */
    @Override
    public void connectionLost(Throwable cause) {
        log.warn("MQTT连接丢失，正在尝试重连...", cause);
    }

    /**
     * 消息发送成功回调
     */
    @Override
    public void deliveryComplete(IMqttDeliveryToken token) {
        try {
            log.debug("MQTT消息发送成功: topic={}", token.getTopics()[0]);
        } catch (MqttException e) {
            log.error("获取发送成功Topic失败", e);
        }
    }

    /**
     * 发布消息到指定Topic
     */
    public void publish(String topic, String payload, int qos) throws MqttException {
        if (mqttClient == null || !mqttClient.isConnected()) {
            throw new MqttException(MqttException.REASON_CODE_CLIENT_NOT_CONNECTED);
        }

        MqttMessage message = new MqttMessage(payload.getBytes());
        message.setQos(qos);
        message.setRetained(false);
        mqttClient.publish(topic, message);
        log.info("MQTT消息发布成功: topic={}, qos={}", topic, qos);
    }
}
```

```java
package com.mqtt.cloud.controller;

import com.baomidou.mybatisplus.core.metadata.IPage;

import io.swagger.v3.oas.annotations.tags.Tag;

@RestController
@RequestMapping("/devices")

        IPage<Device> page = deviceService.getDevices(userId, dto);
        return Result.success(page);
    }

    /**
     * 获取设备详情
     */
    @GetMapping("/{deviceId}")
    @Operation(summary = "获取设备详情", description = "根据设备ID获取详细信息")

        return Result.success(device);

     * 更新设备
     */
    @PutMapping("/{deviceId}")
    @Operation(summary = "更新设备", description = "更新设备信息")
    public Result<Device> updateDevice(@PathVariable Long deviceId,
                                       @Valid @RequestBody UpdateDeviceDTO dto) {
        checkDeviceOwnership(deviceId);
        Device device = deviceService.updateDevice(deviceId, dto);
        return Result.success(device);
    }

    /**
     * 删除设备
     */
    @DeleteMapping("/{deviceId}")
    @Operation(summary = "删除设备", description = "逻辑删除设备")

        checkDeviceOwnership(deviceId);
        deviceService.deleteDevice(deviceId);
        return Result.success();
    }

        List<Device> devices = deviceService.getOnlineDevices();
        return Result.success(devices);
    }

    @GetMapping("/{deviceId}/status")

        checkDeviceOwnership(deviceId);
        List<DeviceStatus> history = deviceService.getBaseMapper()
                .selectList(new LambdaQueryWrapper<DeviceStatus>()
                        .eq(DeviceStatus::getDeviceId, deviceId)
                        .orderByDesc(DeviceStatus::getTimestamp));

    /**
     * 获取当前登录用户ID

    private Long getCurrentUserId() {
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        if (authentication == null || !(authentication.getPrincipal() instanceof UserPrincipal)) {
            throw new BusinessException(ResultCode.UNAUTHORIZED);
        }
        return ((UserPrincipal) authentication.getPrincipal()).getUserId();
    }

    /**
     * 校验设备所有权（防止越权访问）
     */
    private void checkDeviceOwnership(Long deviceId) {
        Long userId = getCurrentUserId();
        Device device = deviceService.getById(deviceId);
        if (device == null) {
            throw new BusinessException(ResultCode.DEVICE_NOT_FOUND);
        }
        // ADMIN可访问所有设备，其他用户只能访问自己的
        String role = ((UserPrincipal) SecurityContextHolder.getContext()
                .getAuthentication().getPrincipal()).getRole();
        if (!"ADMIN".equals(role) && !device.getOwnerId().equals(userId)) {
            throw new BusinessException(ResultCode.DEVICE_NOT_OWNED);
        }
    }
}
```
