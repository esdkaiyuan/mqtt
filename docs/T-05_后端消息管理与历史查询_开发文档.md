## 2. 数据存储架构设计

### 2.1 message与history_record的关系

```
设备上报数据流：
  ┌──────────┐    MQTT消息    ┌──────────────┐
  │ MQTT设备  │ ─────────────► │ MQTT Broker  │ (EMQX)
  │ (传感器)  │               │              │
  └──────────┘               └──────┬───────┘
                                    │ 转发
                                    ▼
                          ┌─────────────────┐
                          │ MqttMessageHandler│ (后端)
                          │ ┌───────────────┐│
                          │ │ 1. 更新设备   ││
                          │ │    lastSeen   ││
                          │ │ 2. 存储message ││
                          │ │    (原始MQTT)  ││
                          │ │ 3. 存储       ││
                          │ │    history     ││
                          │ └───────────────┘│
                          └─────────────────┘
                                    │
                    ┌───────────────┼───────────────┐
                    ▼               ▼               ▼
              ┌───────────┐  ┌───────────┐  ┌───────────┐
              │ message表  │  │history_    │  │ device表   │
              │ (原始消息) │  │record表    │  │ (状态更新) │
              │ QoS/方向/  │  │ (业务数据) │  │           │
              │ 发送接收时间│  │ JSON载荷   │  │           │
              └───────────┘  └───────────┘  └───────────┘
```

**message表 vs history_record表职责分离：**

| 维度 | message表 | history_record表 |
|------|-----------|------------------|
| **存储内容** | 原始MQTT消息（Topic、QoS、方向、发送/接收时间） | 经过处理的业务数据（JSON格式的传感器读数） |
| **写入触发** | 所有MQTT消息（含平台发送的命令） | 仅设备上报的数据消息 |
| **保留期限** | 90天 | 365天 |
| **查询用途** | 消息审计、消息监控、通信排查 | 历史趋势分析、数据可视化、报表 |
| **数据量** | 高频（每次通信都记录） | 中频（仅记录有效业务数据） |
| **数据结构** | payload为TEXT（原始字符串） | payload为JSON（结构化数据） |

**数据流转规则：**
1. 设备上报数据消息 → 存入message表（direction=SUBSCRIBE）+ 存入history_record表
2. 平台下发命令 → 存入message表（direction=PUBLISH），不入history_record
3. 外部系统通过API发布消息 → 存入message表（direction=PUBLISH），若关联设备则更新设备状态

### 2.2 数据保留策略

| 表名 | 保留期限 | 归档/清理策略 | 触发条件 |
|------|----------|---------------|----------|
| message | 90天 | 超过90天的记录由定时任务归档或删除 | 每天凌晨执行 |
| history_record | 365天 | 超过365天的记录归档到冷存储 | 每周执行 |
| device_status_history | 180天 | 超过180天的记录删除 | 每周执行 |

```java
package com.mqtt.cloud.entity;

import com.baomidou.mybatisplus.annotation.*;
import lombok.Data;
import lombok.EqualsAndHashCode;

import java.time.LocalDateTime;

/**
 * 消息实体
 * 对应数据库表：message
 * 存储所有MQTT消息的收发记录（原始消息）
 */
@Data
@EqualsAndHashCode(callSuper = false)
@TableName("message")
public class Message {

    /** 主键ID */
    @TableId(value = "id", type = IdType.AUTO)
    private Long id;

    /** MQTT Topic */
    @TableField("topic")
    private String topic;

    /** 消息方向：PUBLISH=平台发布，SUBSCRIBE=设备上报 */
    @TableField("direction")
    private String direction;

    /** 消息载荷内容（原始字符串） */
    @TableField("payload")
    private String payload;

    /** QoS等级：0/1/2 */
    @TableField("qos")
    private Integer qos;

    /** 关联设备ID（可为空，外部系统消息无关联设备） */
    @TableField("device_id")
    private Long deviceId;

    /** 消息发送时间（发送方时间） */
    @TableField("sent_at")
    private LocalDateTime sentAt;

    /** 消息接收时间（服务端入库时间） */
    @TableField(value = "received_at", fill = FieldFill.INSERT)
    private LocalDateTime receivedAt;

    /** 创建时间 */
    @TableField(value = "created_at", fill = FieldFill.INSERT)
    private LocalDateTime createdAt;
}
```

```java
package com.mqtt.cloud.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.core.metadata.IPage;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.mqtt.cloud.common.ResultCode;
import com.mqtt.cloud.common.exception.BusinessException;
import com.mqtt.cloud.dto.request.MessageQueryDTO;
import com.mqtt.cloud.dto.request.PublishMessageDTO;
import com.mqtt.cloud.entity.Device;
import com.mqtt.cloud.entity.Message;
import com.mqtt.cloud.mapper.MessageMapper;
import com.mqtt.cloud.mqtt.MqttMessageHandler;
import com.mqtt.cloud.service.DeviceService;
import com.mqtt.cloud.service.MessageService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * 消息服务实现类
 */
@Service
@RequiredArgsConstructor
public class MessageServiceImpl extends ServiceImpl<MessageMapper, Message> implements MessageService {

    private final MqttMessageHandler mqttMessageHandler;
    private final DeviceService deviceService;

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Message publishMessage(PublishMessageDTO dto, Long userId) {
        // 1. 校验设备存在且属于当前用户（如果提供了deviceId）
        if (dto.getDeviceId() != null) {
            Device device = deviceService.getById(dto.getDeviceId());
            if (device == null || !device.getOwnerId().equals(userId)) {
                throw new BusinessException(ResultCode.DEVICE_NOT_FOUND);
            }
        }

        // 2. 通过MQTT Broker发送消息
        try {
            mqttMessageHandler.publish(dto.getTopic(), dto.getPayload(), dto.getQos());
        } catch (Exception e) {
            throw new BusinessException("消息发送失败: " + e.getMessage());
        }

        // 3. 记录消息到数据库
        Message message = new Message();
        message.setTopic(dto.getTopic());
        message.setDirection("PUBLISH");  // 平台发布
        message.setPayload(dto.getPayload());
        message.setQos(dto.getQos());
        message.setDeviceId(dto.getDeviceId());
        message.setSentAt(java.time.LocalDateTime.now());
        this.save(message);

        return message;
    }

    @Override
    public IPage<Message> getMessages(MessageQueryDTO dto, Long userId) {
        Page<Message> page = new Page<>(dto.getPageNum(), dto.getPageSize());
        return this.baseMapper.pageQuery(page, dto, userId);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void saveReceivedMessage(Long deviceId, String topic, String payload, Integer qos) {
        Message message = new Message();
        message.setTopic(topic);
        message.setDirection("SUBSCRIBE");  // 设备上报
        message.setPayload(payload);
        message.setQos(qos);
        message.setDeviceId(deviceId);
        message.setSentAt(java.time.LocalDateTime.now());
        this.save(message);
    }
}
```

```java
import org.springframework.security.core.Authentication;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.web.bind.annotation.*;

/**
 * 消息管理控制器
 */
@RestController
@RequestMapping("/messages")
@Tag(name = "消息管理", description = "消息发布、消息查询")
public class MessageController {

    private final MessageService messageService;

    public MessageController(MessageService messageService) {
        this.messageService = messageService;
    }

    /**
     * 发布MQTT消息
     */
    @PostMapping("/publish")
    @Operation(summary = "发布MQTT消息", description = "向指定Topic发布消息，通过MQTT Broker发送到设备")

     * 查询消息列表（分页+过滤）
```
