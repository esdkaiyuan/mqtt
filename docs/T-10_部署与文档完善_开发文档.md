### 2.1 Docker Compose配置

**文件路径：** `docker/docker-compose.yml`

```yaml
version: '3.8'

services:
  # ========== MySQL数据库 ==========
  mysql:
    image: mysql:8.0
    container_name: mqtt-mysql
    restart: always
    environment:
      MYSQL_ROOT_PASSWORD: ${DB_ROOT_PASSWORD:-root_password}
      MYSQL_DATABASE: ${DB_NAME:-mqtt_cloud}
    ports:
      - "${DB_PORT:-3306}:3306"
    volumes:
      - mysql_data:/var/lib/mysql
      - ./scripts/init-mysql.sql:/docker-entrypoint-initdb.d/01-init.sql
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s
    networks:
      - mqtt-network

  # ========== Redis缓存 ==========
  redis:
    image: redis:7-alpine
    container_name: mqtt-redis
    restart: always
    command: redis-server --requirepass ${REDIS_PASSWORD:-}
    ports:
      - "${REDIS_PORT:-6379}:6379"
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - mqtt-network

  # ========== EMQX MQTT Broker ==========
  emqx:
    image: emqx/emqx:5.0
    container_name: mqtt-emqx
    restart: always
    ports:
      - "${MQTT_PORT:-1883}:1883"       # MQTT协议端口
      - "${MQTT_WS_PORT:-8083}:8083"     # WebSocket端口（前端MQTT.js连接）
      - "${EMQX_DASHBOARD_PORT:-18083}:18083"  # EMQX管理后台
    environment:
      - EMQX_NAME=emqx
      - EMQX_HOST=127.0.0.1
      - EMQX_ZONES__DEFAULT__LISTENERS__TCP__DEFAULT__BIND=0.0.0.0:1883
      - EMQX_ZONES__DEFAULT__LISTENERS__WS__DEFAULT__BIND=0.0.0.0:8083
      - EMQX_AUTH__USER__1__USERNAME=${MQTT_USERNAME:-admin}
      - EMQX_AUTH__USER__1__PASSWORD=${MQTT_PASSWORD:-public}
    volumes:
      - emqx_data:/opt/emqx/data
      - emqx_log:/opt/emqx/log
    healthcheck:
      test: ["CMD", "emqx", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s
    networks:
      - mqtt-network
    depends_on:
      - mysql
      - redis

  # ========== 后端服务 ==========
  backend:
    build:
      context: ../backend
      dockerfile: Dockerfile.backend
    container_name: mqtt-backend
    restart: always
    ports:
      - "${BACKEND_PORT:-8080}:8080"
    environment:
      # 数据库配置
      DB_HOST: mysql
      DB_PORT: 3306
      DB_USERNAME: ${DB_USERNAME:-root}
      DB_PASSWORD: ${DB_ROOT_PASSWORD:-root_password}
      DB_NAME: ${DB_NAME:-mqtt_cloud}
      # Redis配置
      REDIS_HOST: redis
      REDIS_PORT: 6379
      REDIS_PASSWORD: ${REDIS_PASSWORD:-}
      # MQTT配置
      MQTT_BROKER_HOST: emqx
      MQTT_BROKER_PORT: 1883
      MQTT_USERNAME: ${MQTT_USERNAME:-admin}
      MQTT_PASSWORD: ${MQTT_PASSWORD:-public}
      # JWT配置（生产环境必须更换）
      JWT_SECRET: ${JWT_SECRET:-mqttCloudPlatformSecretKey2026}
      # 服务器配置
      SERVER_PORT: 8080
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8080/api/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 60s
    networks:
      - mqtt-network
    depends_on:
      mysql:
        condition: service_healthy
      redis:
        condition: service_healthy
      emqx:
        condition: service_healthy

  # ========== 前端服务 ==========
  frontend:
    build:
      context: ../frontend
      dockerfile: Dockerfile.frontend
      args:
        VITE_API_URL: /api
    container_name: mqtt-frontend
    restart: always
    ports:
      - "${FRONTEND_PORT:-80}:80"
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost"]
      interval: 30s
      timeout: 10s
      retries: 3
    networks:
      - mqtt-network
    depends_on:
      backend:
        condition: service_healthy

# ========== 数据卷 ==========
volumes:
  mysql_data:
    driver: local
  redis_data:
    driver: local
  emqx_data:
    driver: local
  emqx_log:
    driver: local

# ========== 网络 ==========

### 2.2 后端Dockerfile

**文件路径：** `docker/Dockerfile.backend`

```dockerfile
# ========== 构建阶段 ==========
FROM maven:3.9-eclipse-temurin-17-alpine AS builder
WORKDIR /build

# 复制pom.xml并下载依赖（利用Docker缓存）
COPY backend/pom.xml .
COPY backend/.m2.settings.xml .m2/settings.xml 2>/dev/null || true
RUN mvn dependency:go-offline -B

# 复制源代码并打包
COPY backend/src ./src
COPY backend/pom.xml .
RUN mvn clean package -DskipTests -B

# ========== 运行阶段 ==========
FROM eclipse-temurin:17-jre-alpine
WORKDIR /app

# 安装curl用于健康检查
RUN apk add --no-cache curl

# 复制构建好的jar包
COPY --from=builder /build/target/*.jar app.jar

# 暴露端口
EXPOSE 8080

# 启动命令
ENTRYPOINT ["java", "-jar", "app.jar"]
```

### 2.4 Nginx配置

**文件路径：** `docker/nginx.conf`

```nginx
worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for"';

    access_log /var/log/nginx/access.log main;

    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;

    # Gzip压缩
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/xml text/javascript application/x-javascript application/xml+rss application/javascript application/json;

    # 前端服务
    server {
        listen 80;
        server_name _;
        root /usr/share/nginx/html;
        index index.html;

        # 静态资源缓存
        location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
            expires 1y;
            add_header Cache-Control "public, immutable";
        }

        # SPA路由：所有非API请求都返回index.html
        location / {
            try_files $uri $uri/ /index.html;
        }

        # 代理API请求到后端
        location /api {
            proxy_pass http://backend:8080;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # 超时设置
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;
        }

        # 健康检查
        location /health {
            access_log off;
            return 200 "healthy\n";
        }
    }
}
```

### 2.6 部署脚本

**文件路径：** `scripts/deploy.sh`

> 说明：下方为初版设计稿（v1.0）。**当前实现已演进**：前后端均在容器内构建，本机无需 Maven/Node；脚本以 `--env-file "$ROOT_DIR/.env" -f "$ROOT_DIR/docker/docker-compose.yml"` 方式调用 compose，并自动生成缺失的 `JWT_SECRET`。以仓库中的 `scripts/deploy.sh` 为准。

```bash
#!/bin/bash
set -e

echo "=========================================="
echo "  MQTT云平台部署脚本 v1.0"
echo "=========================================="

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# 1. 检查依赖
echo -e "\n${YELLOW}[1/6] 检查依赖...${NC}"
command -v docker >/dev/null 2>&1 || { echo -e "${RED}错误：Docker未安装${NC}"; exit 1; }
command -v docker-compose >/dev/null 2>&1 || { echo -e "${RED}错误：Docker Compose未安装${NC}"; exit 1; }
command -v mvn >/dev/null 2>&1 || { echo -e "${RED}错误：Maven未安装${NC}"; exit 1; }
command -v node >/dev/null 2>&1 || { echo -e "${RED}错误：Node.js未安装${NC}"; exit 1; }
echo -e "${GREEN}✓ 依赖检查通过${NC}"

# 2. 读取环境变量
echo -e "\n${YELLOW}[2/6] 读取环境配置...${NC}"
if [ -f .env ]; then
    export $(cat .env | xargs)
    echo -e "${GREEN}✓ 已加载.env文件${NC}"
else
    if [ -f .env.example ]; then
        cp .env.example .env
        echo -e "${YELLOW}⚠ 未找到.env文件，已从.env.example创建，请修改配置${NC}"
    fi
fi

# 3. 构建后端
echo -e "\n${YELLOW}[3/6] 构建后端...${NC}"
cd backend
mvn clean package -DskipTests -q
if [ $? -ne 0 ]; then
    echo -e "${RED}✗ 后端构建失败${NC}"
    exit 1
fi
echo -e "${GREEN}✓ 后端构建成功${NC}"
cd ..

# 4. 构建前端
echo -e "\n${YELLOW}[4/6] 构建前端...${NC}"
cd frontend
npm install --silent
npm run build
if [ $? -ne 0 ]; then
    echo -e "${RED}✗ 前端构建失败${NC}"
    exit 1
fi
echo -e "${GREEN}✓ 前端构建成功${NC}"
cd ..

# 5. 启动Docker Compose
echo -e "\n${YELLOW}[5/6] 启动Docker Compose...${NC}"
cd docker
docker-compose down -v 2>/dev/null || true  # 清理旧容器
docker-compose up -d --build
echo -e "${GREEN}✓ Docker Compose启动成功${NC}"

# 6. 等待服务启动
echo -e "\n${YELLOW}[6/6] 等待服务启动...${NC}"
echo "等待MySQL启动（约30秒）..."
sleep 30

# 7. 执行健康检查
echo -e "\n${YELLOW}[7/7] 执行健康检查...${NC}"
cd ..
bash scripts/health-check.sh

echo ""
echo "=========================================="
echo -e "${GREEN}  ✓ 部署完成！${NC}"
echo "=========================================="
echo ""
echo "访问地址："
echo "  前端页面：http://localhost"
echo "  后端API：  http://localhost:8080/api"
echo "  API文档：  http://localhost:8080/api/swagger-ui.html"
echo "  EMQX后台： http://localhost:18083"
echo ""
echo "默认账号：admin / admin123"
echo ""
```

### 2.7 健康检查脚本

**文件路径：** `scripts/health-check.sh`

```bash
#!/bin/bash

echo "=========================================="
echo "  MQTT云平台健康检查"
echo "=========================================="

PASS=0
FAIL=0

# 检查函数
check_service() {
    local name=$1
    local url=$2
    local expected=$3

    echo -n "检查 $name ... "

    if [ -z "$url" ]; then
        # Docker容器健康检查
        local status=$(docker inspect --format='{{.State.Health.Status}}' $1 2>/dev/null)
        if [ "$status" = "healthy" ]; then
            echo -e "\033[0;32m✓ 正常\033[0m"
            ((PASS++))
        else
            echo -e "\033[0;31m✗ 异常（状态: ${status:-unknown}）\033[0m"
            ((FAIL++))
        fi
    else
        # HTTP健康检查
        local response=$(curl -s -o /dev/null -w "%{http_code}" --max-time 5 "$url" 2>/dev/null)
        if [ "$response" = "$expected" ]; then
            echo -e "\033[0;32m✓ 正常（HTTP $response）\033[0m"
            ((PASS++))
        else
            echo -e "\033[0;31m✗ 异常（HTTP $response，期望 $expected）\033[0m"
            ((FAIL++))
        fi
    fi
}

# 1. 检查MySQL
check_service "mqtt-mysql" "" ""

# 2. 检查Redis
check_service "mqtt-redis" "" ""

# 3. 检查EMQX
check_service "mqtt-emqx" "http://localhost:18083/api/v5/health" "200"

# 4. 检查后端API
check_service "后端API" "http://localhost:8080/api/health" "200"

# 5. 检查前端
check_service "前端" "http://localhost/" "200"

# 6. 检查数据库表
echo -n "检查数据库表 ... "
DB_TABLES=$(docker exec mqtt-mysql mysql -u root -proot_password mqtt_cloud -e "SHOW TABLES;" 2>/dev/null | grep -v "Tables_in" | wc -l)
if [ "$DB_TABLES" = "5" ]; then
    echo -e "\033[0;32m✓ 正常（$DB_TABLES 个表）\033[0m"
    ((PASS++))
else
    echo -e "\033[0;31m✗ 异常（期望5个表，实际 $DB_TABLES 个）\033[0m"
    ((FAIL++))
fi

# 7. 检查默认管理员
echo -n "检查默认管理员 ... "
ADMIN_EXISTS=$(docker exec mqtt-mysql mysql -u root -proot_password mqtt_cloud -e "SELECT COUNT(*) FROM sys_user WHERE username='admin';" 2>/dev/null | grep -v "COUNT" | xargs)
if [ "$ADMIN_EXISTS" = "1" ]; then
    echo -e "\033[0;32m✓ 正常\033[0m"
    ((PASS++))
else
    echo -e "\033[0;31m✗ 异常（admin用户不存在）\033[0m"
    ((FAIL++))
fi

# 汇总结果
echo ""
echo "=========================================="
echo "  健康检查结果"
echo "=========================================="
echo -e "通过：\033[0;32m$PASS\033[0m"
echo -e "失败：\033[0;31m$FAIL\033[0m"
echo ""

if [ $FAIL -eq 0 ]; then
    echo -e "\033[0;32m✓ 所有服务正常\033[0m"
    exit 0
else
    echo -e "\033[0;31m✗ 存在异常服务，请检查日志\033[0m"
    echo ""
    echo "查看日志："
    echo "  docker logs mqtt-mysql"
    echo "  docker logs mqtt-redis"
    echo "  docker logs mqtt-emqx"
    echo "  docker logs mqtt-backend"
    echo "  docker logs mqtt-frontend"
    exit 1
fi
```

## 4. 常见问题

### Q1: 端口冲突怎么办？
A: 修改`.env`文件中的对应端口号，如将`FRONTEND_PORT=80`改为`FRONTEND_PORT=8081`

### Q2: 数据库连接失败？
A: 检查MySQL容器是否正常启动：`docker logs mqtt-mysql`。确认环境变量中的密码与MySQL初始化一致。

### Q3: 前端无法访问？
A: 检查前端容器是否正常启动：`docker logs mqtt-frontend`。确认80端口未被占用。

### Q4: MQTT连接失败？
A: 确认EMQX容器运行正常：`docker logs mqtt-emqx`。检查MQTT配置中的用户名密码是否正确。

### Q5: 如何备份数据？
A:
```bash
# 备份MySQL
docker exec mqtt-mysql mysqldump -u root -proot_password mqtt_cloud > backup.sql

# 备份Redis
docker exec mqtt-redis redis-cli BGSAVE
```

### Q6: 如何升级版本？
A:
```bash
# 1. 拉取最新代码
git pull

# 2. 重新构建并启动
bash scripts/deploy.sh
```

## 5. 生产环境建议

1. **更换JWT密钥：** 修改`.env`中的`JWT_SECRET`为强随机字符串

## 9. 常见问题

**Q: 如何修改部署后的端口？**
A: 修改`.env`文件中的对应端口变量，然后重新执行`bash scripts/deploy.sh`。

**Q: 如何查看服务日志？**
A: 使用`docker logs <容器名>`，如`docker logs mqtt-backend`。

**Q: 如何停止服务？**
A: 在仓库根目录执行 `docker compose --env-file .env -f docker/docker-compose.yml down`，数据卷保留。如需删除数据卷：`docker compose --env-file .env -f docker/docker-compose.yml down -v`。

**Q: 生产环境需要哪些额外配置？**
A: 更换JWT_SECRET、启用HTTPS（Nginx配置SSL）、配置防火墙、设置日志轮转、配置自动备份。

**Q: 如何扩展服务？**
A: Docker Compose支持scale：`docker compose --env-file .env -f docker/docker-compose.yml up -d --scale backend=3`（后端3个实例）。需配合负载均衡器。
