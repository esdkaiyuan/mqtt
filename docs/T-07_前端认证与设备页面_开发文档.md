  username: '',
  password: ''

async function handleLogin() {
  // 清除之前的错误

  } finally {
    loading.value = false
  }
}

<script setup>
import { ref, reactive } from 'vue'
import { useRouter } from 'vue-router'
import { useAuthStore } from '@/stores/auth'
import { ElMessage } from 'element-plus'

const router = useRouter()
const authStore = useAuthStore()

// 表单数据
const form = reactive({
  username: '',
  email: '',
  phone: '',
  password: '',
  confirmPassword: ''
})

// 校验错误
const errors = reactive({
  username: '',
  email: '',
  phone: '',
  password: '',
  confirmPassword: ''
})

// 状态
const loading = ref(false)
const submitError = ref('')

// 表单校验
function validateForm() {
  let isValid = true
  // 重置错误
  Object.keys(errors).forEach(key => errors[key] = '')

  // 用户名校验
  if (!form.username) {
    errors.username = '请输入用户名'
    isValid = false
  } else if (form.username.length < 3 || form.username.length > 50) {
    errors.username = '用户名长度3-50个字符'
    isValid = false
  } else if (!/^[a-zA-Z0-9_]+$/.test(form.username)) {
    errors.username = '用户名只能包含字母、数字、下划线'
    isValid = false
  }

  // 邮箱校验（如果填写了）
  if (form.email) {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
    if (!emailRegex.test(form.email)) {
      errors.email = '邮箱格式不正确'
      isValid = false
    }
  }

  // 手机号校验（如果填写了）
  if (form.phone) {
    const phoneRegex = /^1[3-9]\d{9}$/
    if (!phoneRegex.test(form.phone)) {
      errors.phone = '手机号格式不正确'
      isValid = false
    }
  }

  // 密码校验
  if (!form.password) {
    errors.password = '请输入密码'
    isValid = false
  } else if (form.password.length < 6 || form.password.length > 100) {
    errors.password = '密码长度6-100个字符'
    isValid = false
  }

  // 确认密码校验
  if (form.password !== form.confirmPassword) {
    errors.confirmPassword = '两次输入的密码不一致'
    isValid = false
  }

  return isValid
}

// 注册处理
async function handleRegister() {
  submitError.value = ''

  // 前端校验
  if (!validateForm()) {
    return
  }

  loading.value = true
  try {
    await authStore.register({
      username: form.username,
      email: form.email || undefined,
      phone: form.phone || undefined,
      password: form.password
    })
    ElMessage.success('注册成功，请登录')
  } catch (error) {
    submitError.value = error.response?.data?.message || '注册失败，请稍后重试'
  } finally {
    loading.value = false
  }
}
</script>

  // 重置表单

    else createForm[key] = ''
  })
  showCreateDialog.value = true

async function handleCreate() {
  // 表单校验
  if (!createForm.deviceName || !createForm.deviceKey || !createForm.deviceType || !createForm.topic) {
    ElMessage.warning('请填写所有必填项')
    return

    await deviceStore.createDevice(createForm)
    ElMessage.success('设备创建成功')

    loadDevices()
  } catch (error) {
    // 错误已在store中处理
  } finally {
    createLoading.value = false
  }
}

    await deviceStore.updateDevice(editForm.id, editForm)

    showEditDialog.value = false
    loadDevices()
  } catch (error) {
    // 错误已在store中处理
  } finally {
    editLoading.value = false
  }
}

        confirmButtonText: '删除',
        cancelButtonText: '取消',
        type: 'warning'
      }
    )

    loadDevices()
  } catch (error) {
    // 用户取消删除
  }
}
