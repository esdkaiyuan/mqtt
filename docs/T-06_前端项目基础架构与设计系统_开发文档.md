### 2.1 色板体系

#### 2.1.1 基础色（黑白灰色系）

| CSS变量名 | 色值 | 用途 | 示例 |
|-----------|------|------|------|
| `--color-white` | #FFFFFF | 页面背景、卡片背景 | 主内容区背景 |
| `--color-gray-light` | #F5F5F5 | 侧边栏背景、hover状态 | 侧边栏底色 |
| `--color-gray-border` | #E0E0E0 | 边框、分割线 | 卡片边框、输入框边框 |
| `--color-gray-text` | #666666 | 次要文字、placeholder | 描述文字、辅助说明 |
| `--color-gray-dark` | #333333 | 主标题、重要文字 | 页面标题、按钮文字 |

#### 2.1.2 点缀色（特定场景使用）

| CSS变量名 | 色值 | 用途 | 使用场景 |
|-----------|------|------|----------|
| `--color-primary` | #1890FF | 主要交互元素 | 主操作按钮（登录、创建设备、发布消息）、链接、当前导航高亮 |
| `--color-success` | #52C41A | 成功状态 | 在线状态标识、成功提示、操作成功反馈 |
| `--color-warning` | #FAAD14 | 警告状态 | 警告提示、设备即将超时提醒 |
| `--color-danger` | #FF4D4F | 危险/错误状态 | 离线状态、删除按钮、错误提示、危险操作按钮 |

#### 2.1.3 使用原则

- **默认状态：** 所有界面元素默认使用黑白灰色系
- **点缀色触发条件：** 仅在以下场景使用点缀色
  - 状态文字（在线=绿色、离线=红色、警告=橙色）
  - 交互按钮（主按钮=蓝色、危险按钮=红色）
  - 反馈提示（成功=绿色、错误=红色、警告=橙色）
  - 当前激活的导航项（蓝色高亮）
- **禁止：** 背景色、卡片背景、大面积装饰性元素使用点缀色

### 2.2 字体体系

| CSS变量 | 值 | 用途 |
|---------|-----|------|
| `--font-family` | 'Inter', -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif | 全局字体 |
| `--font-size-2xl` | 32px | 页面大标题 |
| `--font-size-xl` | 24px | 卡片标题 |
| `--font-size-lg` | 20px | 区块标题 |
| `--font-size-md` | 16px | 正文内容 |
| `--font-size-sm` | 14px | 次要文字、输入框文字 |
| `--font-size-xs` | 12px | 辅助说明、时间戳 |

### 2.3 间距体系

| CSS变量 | 值 | 用途 |
|---------|-----|------|
| `--spacing-xs` | 4px | 图标与文字间距 |
| `--spacing-sm` | 8px | 组件内小间距 |
| `--spacing-md` | 16px | 卡片内边距、列表项间距 |
| `--spacing-lg` | 24px | 区块间距 |
| `--spacing-xl` | 32px | 页面区块大间距 |

### 2.4 阴影与圆角

| CSS变量 | 值 | 用途 |
|---------|-----|------|
| `--shadow-sm` | 0 2px 8px rgba(0,0,0,0.06) | 卡片轻微阴影 |
| `--shadow-md` | 0 4px 12px rgba(0,0,0,0.08) | 弹出框阴影 |
| `--shadow-lg` | 0 8px 24px rgba(0,0,0,0.12) | 模态框阴影 |
| `--border-radius` | 6px | 卡片、按钮圆角 |
| `--border-color` | #E0E0E0 | 边框颜色 |

### 2.5 极简风格规则

1. **无装饰性元素：** 禁止渐变背景、禁止背景图片、禁止装饰性图标
2. **留白充足：** 卡片之间、区块之间使用--spacing-lg或--spacing-xl
3. **线条简洁：** 使用1px #E0E0E0边框作为视觉分割，替代多余的空格
4. **对齐严格：** 所有元素左对齐或居中对齐，禁止错位
5. **字体克制：** 仅使用3种字号层级（标题-md、正文-sm、辅助-xs）

#### 3.4.3 SVG图标注册（src/assets/svg/index.js）

```javascript
// 导入所有SVG图标
const icons = import.meta.glob('./icons/*.svg', { query: '?raw', import: 'default' })

// 注册为Vue组件（在App.vue中使用）
export function registerIcons(app) {
  const svgSprite = document.createElement('div')
  svgSprite.style.display = 'none'
  svgSprite.id = 'svg-sprite'
  document.body.appendChild(svgSprite)

  // 异步加载所有SVG并插入到sprite中
  Object.keys(icons).forEach(path => {
    const name = path.match(/\/([^/]+)\.svg$/)?.[1]
    if (name) {
      icons[path]().then(svgContent => {
        const symbol = document.createElementNS('http://www.w3.org/2000/svg', 'symbol')
        symbol.id = `icon-${name}`
        symbol.setAttribute('viewBox', '0 0 24 24')
        symbol.innerHTML = svgContent.trim().replace(/<svg[^>]*>/, '').replace(/<\/svg>/, '')
        svgSprite.appendChild(symbol)
      })
    }
  })
}
```

```javascript
import axios from 'axios'
import { ElMessage } from 'element-plus'
import { useAuthStore } from '@/stores/auth'

const api = axios.create({
  baseURL: '/api',
  timeout: 10000
})

// 请求拦截器：自动添加Token
api.interceptors.request.use(
  (config) => {
    const authStore = useAuthStore()
    if (authStore.token) {
      config.headers.Authorization = `Bearer ${authStore.token}`
    }
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// 响应拦截器：统一处理响应和错误
api.interceptors.response.use(
  (response) => {
    // 直接返回data字段（Result中的data）
    return response.data
  },
  (error) => {
    // 错误处理
    const errorData = error.response?.data
    if (errorData) {
      // 显示错误消息
      ElMessage.error(errorData.message || '请求失败')
    } else {
      ElMessage.error('网络错误，请稍后重试')
    }

    // Token过期或无效时跳转到登录页
    if (error.response?.status === 401) {
      const authStore = useAuthStore()
      authStore.logout()
    }

    return Promise.reject(error)
  }
)
```

### 3.8 Pinia状态管理

#### 3.8.1 src/stores/auth.js

```javascript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import api from '@/api/axios'
import router from '@/router'

export const useAuthStore = defineStore('auth', () => {
  // ========== State ==========
  const token = ref(localStorage.getItem('token') || '')
  const user = ref(JSON.parse(localStorage.getItem('user') || 'null'))

  // ========== Getters ==========
  const isLoggedIn = computed(() => !!token.value)

  // ========== Actions ==========

  /**
   * 用户登录
   */
  async function login(loginDTO) {
    try {
      const response = await api.post('/auth/login', loginDTO)
      // 登录成功，保存token和用户信息
      token.value = response.data.token
      user.value = {
        id: response.data.id,
        username: response.data.username,
        email: response.data.email,
        phone: response.data.phone,
        role: response.data.role,
        status: response.data.status
      }

      // 持久化到localStorage
      localStorage.setItem('token', token.value)
      localStorage.setItem('user', JSON.stringify(user.value))

      // 跳转到仪表盘
      router.push('/dashboard')
    } catch (error) {
      throw error
    }
  }

  /**
   * 用户注册
   */
  async function register(registerDTO) {
    try {
      const response = await api.post('/auth/register', registerDTO)
      // 注册成功，跳转到登录页
      router.push('/login')
      return response
    } catch (error) {
      throw error
    }
  }

  /**
   * 用户登出
   */
  async function logout() {
    try {
      // 调用后端登出接口（将Token加入黑名单）
      await api.post('/auth/logout')
    } catch (error) {
      console.error('登出失败:', error)
    } finally {
      // 无论成功与否，都清除本地状态
      token.value = ''
      user.value = null
      localStorage.removeItem('token')
      localStorage.removeItem('user')

      // 跳转到登录页
      router.push('/login')
    }
  }

  /**
   * 获取当前用户信息
   */
  async function fetchCurrentUser() {
    try {
      const response = await api.get('/auth/current')
      user.value = response.data
      return response.data
    } catch (error) {
      // 获取失败，清除状态
      token.value = ''
      user.value = null
      localStorage.removeItem('token')
      localStorage.removeItem('user')
      throw error
    }
  }

  return {
    token,
    user,
    isLoggedIn,
    login,
    register,
    logout,
    fetchCurrentUser
  }
})
```

```javascript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import api from '@/api/axios'

export const useDeviceStore = defineStore('device', () => {
  // ========== State ==========
  const devices = ref([])
  const onlineDevices = ref([])
  const totalDevices = ref(0)
  const loading = ref(false)

  // ========== Getters ==========
  const onlineCount = computed(() => onlineDevices.value.length)
  const offlineCount = computed(() => totalDevices.value - onlineDevices.value.length)

  // ========== Actions ==========

  /**
   * 分页获取设备列表
   */
  async function fetchDevices(params = {}) {
    loading.value = true
    try {
      const response = await api.get('/devices', { params })
      devices.value = response.data.records || []
      totalDevices.value = response.data.total || 0
      return response.data
    } finally {
      loading.value = false
    }
  }

  /**
   * 创建设备
   */
  async function createDevice(deviceData) {
    const response = await api.post('/devices', deviceData)
    devices.value.push(response.data)
    totalDevices.value++
    return response.data
  }

  /**
   * 更新设备
   */
  async function updateDevice(deviceId, deviceData) {
    const response = await api.put(`/devices/${deviceId}`, deviceData)
    const index = devices.value.findIndex(d => d.id === deviceId)
    if (index !== -1) {
      devices.value[index] = response.data
    }
    return response.data
  }

  /**
   * 删除设备
   */
  async function deleteDevice(deviceId) {
    await api.delete(`/devices/${deviceId}`)
    devices.value = devices.value.filter(d => d.id !== deviceId)
    totalDevices.value--
  }

  /**
   * 获取在线设备
   */
  async function fetchOnlineDevices() {
    try {
      const response = await api.get('/devices/online')
      onlineDevices.value = response.data
    } catch (error) {
      console.error('获取在线设备失败:', error)
    }
  }

  return {
    devices,
    onlineDevices,
    totalDevices,
    loading,
    onlineCount,
    offlineCount,
    fetchDevices,
    createDevice,
    updateDevice,
    deleteDevice,
    fetchOnlineDevices
  }
})
```
