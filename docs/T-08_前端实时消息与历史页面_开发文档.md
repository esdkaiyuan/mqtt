<script setup>
import { ref, onMounted, onUnmounted, nextTick } from 'vue'
import mqtt from 'mqtt'
import { useAuthStore } from '@/stores/auth'
import api from '@/api/axios'
import SvgIcon from '@/components/Icon.vue'
import { ElMessage } from 'element-plus'

const authStore = useAuthStore()

// MQTT连接状态
const mqttConnected = ref(false)
let mqttClient = null

// 消息列表
const messages = ref([])
const messageContainer = ref(null)
const onlineCount = ref(0)

// 发布表单
const publishForm = ref({
  topic: '',
  qos: 0,
  payload: ''
})
const publishLoading = ref(false)

// MQTT连接配置
const MQTT_URL = 'ws://localhost:8083/mqtt'
const MQTT_OPTIONS = {
  clientId: 'frontend_' + Date.now(),
  username: 'admin',
  password: 'public',
  clean: true,
  connectTimeout: 4000,
  reconnectPeriod: 5000
}

// 初始化MQTT连接
function initMqtt() {
  try {
    mqttClient = mqtt.connect(MQTT_URL, MQTT_OPTIONS)

    mqttClient.on('connect', () => {
      mqttConnected.value = true
      console.log('MQTT连接成功')

      // 订阅所有设备数据Topic
      mqttClient.subscribe('device/+/data', { qos: 0 }, (err) => {
        if (err) {
          console.error('订阅失败:', err)
        } else {
          console.log('订阅成功: device/+/data')
        }
      })

      // 订阅设备状态Topic
      mqttClient.subscribe('device/+/status', { qos: 0 })
    })

    mqttClient.on('message', (topic, message) => {
      const payload = message.toString()
      console.log('收到消息:', topic, payload)

      // 从Topic提取deviceKey
      const deviceKey = topic.split('/')[1]

      // 添加到消息列表
      messages.value.push({
        id: Date.now() + Math.random(),
        topic,
        payload,
        direction: 'SUBSCRIBE',
        qos: 0,
        receivedAt: new Date().toISOString()
      })

      // 限制消息数量（最多200条）
      if (messages.value.length > 200) {
        messages.value.shift()
      }

      // 自动滚动到底部
      scrollToBottom()
    })

    mqttClient.on('error', (err) => {
      console.error('MQTT错误:', err)
      mqttConnected.value = false
    })

    mqttClient.on('offline', () => {
      mqttConnected.value = false
    })

    mqttClient.on('reconnect', () => {
      console.log('MQTT重连中...')
      mqttConnected.value = false
    })

  } catch (error) {
    console.error('MQTT连接失败:', error)
    mqttConnected.value = false
  }
}

// 滚动到底部
function scrollToBottom() {
  nextTick(() => {
    if (messageContainer.value) {
      messageContainer.value.scrollTop = messageContainer.value.scrollHeight
    }
  })
}

// 发布消息
async function handlePublish() {
  publishLoading.value = true
  try {
    const response = await api.post('/messages/publish', {
      topic: publishForm.value.topic,
      payload: publishForm.value.payload,
      qos: parseInt(publishForm.value.qos)
    })

    // 添加到消息列表（作为发送的消息）
    messages.value.push({
      id: Date.now(),
      topic: publishForm.value.topic,
      payload: publishForm.value.payload,
      direction: 'PUBLISH',
      qos: parseInt(publishForm.value.qos),
      receivedAt: new Date().toISOString()
    })

    ElMessage.success('消息发布成功')
    scrollToBottom()

    // 清空payload
    publishForm.value.payload = ''
  } catch (error) {
    // 错误已在axios拦截器中显示
  } finally {
    publishLoading.value = false
  }
}

// 清空消息列表
function clearMessages() {
  messages.value = []
}

// 刷新在线设备数量
async function refreshOnlineCount() {
  try {
    const response = await api.get('/devices/online')
    onlineCount.value = response.data?.length || 0
  } catch (error) {
    console.error('获取在线设备失败:', error)
  }
}

// 格式化时间
function formatTime(timeStr) {
  if (!timeStr) return ''
  const date = new Date(timeStr)
  const now = new Date()
  const diff = now - date

  // 今天内的消息显示时间，否则显示日期
  if (diff < 86400000 && date.getDate() === now.getDate()) {
    return date.toLocaleTimeString('zh-CN', { hour: '2-digit', minute: '2-digit', second: '2-digit' })
  }
  return date.toLocaleString('zh-CN', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
}

// 格式化Payload（JSON美化）
function formatPayload(payload) {
  try {
    const obj = JSON.parse(payload)
    return JSON.stringify(obj, null, 2)
  } catch {
    return payload
  }
}

// 页面加载
onMounted(() => {
  initMqtt()
  refreshOnlineCount()
})

// 页面卸载时断开MQTT连接
onUnmounted(() => {
  if (mqttClient) {
    mqttClient.end()
    mqttClient = null
  }
})
</script>

import { ref, reactive, computed, onMounted } from 'vue'
import { useDeviceStore } from '@/stores/device'
import api from '@/api/axios'
import SvgIcon from '@/components/Icon.vue'
import { ElMessage } from 'element-plus'

const deviceStore = useDeviceStore()

// 设备选项
const deviceOptions = computed(() => deviceStore.devices)

// 查询表单
const queryForm = reactive({
  deviceId: '',
  topic: '',
  startTime: '',
  endTime: ''
})

// 查询状态
const hasSearched = ref(false)
const queryLoading = ref(false)
const historyRecords = ref([])
const totalRecords = ref(0)
const currentPage = ref(1)
const pageSize = ref(20)
const totalPages = computed(() => Math.ceil(totalRecords.value / pageSize.value))

// 页面加载时获取设备列表
onMounted(async () => {
  await deviceStore.fetchDevices({ pageNum: 1, pageSize: 100 })
})

// 执行查询
async function handleQuery() {
  queryLoading.value = true
  hasSearched.value = true
  try {
    const params = {
      pageNum: currentPage.value,
      pageSize: pageSize.value
    }

    if (queryForm.deviceId) params.deviceId = queryForm.deviceId
    if (queryForm.topic) params.topic = queryForm.topic
    if (queryForm.startTime) params.startTime = queryForm.startTime
    if (queryForm.endTime) params.endTime = queryForm.endTime

    const response = await api.get('/history', { params })
    historyRecords.value = response.data?.records || []
    totalRecords.value = response.data?.total || 0
  } catch (error) {
    ElMessage.error('查询失败')
  } finally {
    queryLoading.value = false
  }
}

// 重置查询
function handleReset() {
  queryForm.deviceId = ''
  queryForm.topic = ''
  queryForm.startTime = ''
  queryForm.endTime = ''
  hasSearched.value = false
  historyRecords.value = []
  totalRecords.value = 0
  currentPage.value = 1
}

// 分页
function goToPage(page) {
  currentPage.value = page
  handleQuery()
}

// 格式化时间
function formatTime(timeStr) {
  if (!timeStr) return '-'
  return new Date(timeStr).toLocaleString('zh-CN')
}

// 格式化Payload
function formatPayload(payload) {
  if (!payload) return ''
  try {
    const obj = JSON.parse(payload)
    return JSON.stringify(obj, null, 2)
  } catch {
    return payload
  }
}
</script>

## 3. API封装

### 3.1 src/api/message.js

```javascript
import api from './axios'

export const messageApi = {
  /**
   * 发布MQTT消息
   */
  publish: (data) => api.post('/messages/publish', data),

  /**
   * 查询消息列表（分页+过滤）
   * @param {Object} params - { pageNum, pageSize, topic, deviceId, direction, startTime, endTime }
   */
  getList: (params) => api.get('/messages', { params }),

  /**
   * 获取最近消息
   * @param {Number} limit - 数量限制，默认50
   */
  getRecent: (limit = 50) => api.get('/messages/recent', { params: { limit } })
}
```

### 3.2 src/api/history.js

```javascript
import api from './axios'

export const historyApi = {
  /**
   * 查询历史记录（分页+过滤）
   * @param {Object} params - { pageNum, pageSize, deviceId, topic, startTime, endTime }
   */
  query: (params) => api.get('/history', { params }),

  /**
   * 查询指定设备的历史记录
   * @param {Number} deviceId - 设备ID
   * @param {Object} params - { pageNum, pageSize, startTime, endTime }
   */
  getDeviceHistory: (deviceId, params) => api.get(`/history/device/${deviceId}`, { params })
}
```
